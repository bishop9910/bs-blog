3:  * Playlist drawer for the music player.
4:  *
5:  * The full library is the queue by default; picking a row plays that track

10: import { computed, ref } from 'vue'

20: const visible = computed<MusicTrack[]>(() => {
21:   const needle = keyword.value.trim().toLowerCase()
22:   const all = music.tracks.value
23:   if (!needle) return all
24: 
25:   return all.filter((track) =>
26:     `${track.title} ${track.artist} ${track.album}`.toLowerCase().includes(needle),
27:   )
28: })
