157: /** Average of the low bins — the part that reads as "bass" to the eye. */
158: function readLevel(): number {
159:   if (!analyser) return 0

165:   return total / ((bins - 1) * 255)

589:  * Loads the library once per session. The player mounts on every public page,

592: async function load(force = false): Promise<void> {
593:   if (loaded.value && !force) return
594:   if (inflight) return inflight
595: 
596:   loading.value = true
597:   loadError.value = ''
598: 
599:   inflight = getMusicLibrary()
600:     .then((data) => {
601:       library.value = data
602:       loaded.value = true
603: 
604:       const ids = data.tracks.map((track) => track.id)
605:       const known = new Set(ids)
606: 
607:       // A rescan can drop tracks; a stale queue id would stall "next" on a
608:       // missing file, so the queue is rebuilt from what the server just sent.
609:       queue.value = queue.value.length > 0 ? queue.value.filter((id) => known.has(id)) : ids
610:       if (queue.value.length === 0) queue.value = ids
611: 
612:       if (!currentId.value) {
613:         currentId.value = stored.trackId && known.has(stored.trackId) ? stored.trackId : (ids[0] ?? '')
614:       } else if (!known.has(currentId.value)) {
615:         currentId.value = ids[0] ?? ''
616:       }
617: 
618:       // Prepare (but never auto-play) the restored track so the panel shows its
619:       // cover, title and duration before the visitor asks for sound.
620:       if (currentId.value) select(currentId.value, false)
621:       else duration.value = 0
622:     })
623:     .catch((cause: unknown) => {
624:       loadError.value = cause instanceof Error ? cause.message : '音乐库加载失败'
625:     })
626:     .finally(() => {
627:       loading.value = false
628:       inflight = null
