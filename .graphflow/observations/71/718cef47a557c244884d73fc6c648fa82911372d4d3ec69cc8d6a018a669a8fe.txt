230: // rfc3339 renders a timestamp in the format the frontend parses.
231: func rfc3339(t time.Time) string { return t.UTC().Format(time.RFC3339) }
232: 
233: // rfc3339Ptr renders an optional timestamp.
234: func rfc3339Ptr(t *time.Time) *string {
235: 	if t == nil {
236: 		return nil
237: 	}
238: 	s := rfc3339(*t)
239: 	return &s
240: }
