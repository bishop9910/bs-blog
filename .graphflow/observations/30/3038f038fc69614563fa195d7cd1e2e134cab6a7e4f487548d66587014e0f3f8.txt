  72: type QueryValue = string | number | boolean | undefined | null
  73: 
  74: export function buildQuery(params: Record<string, QueryValue>): string {
  75:   const search = new URLSearchParams()
  76:   for (const [key, value] of Object.entries(params)) {
  77:     if (value === undefined || value === null || value === '') continue
  78:     search.set(key, String(value))
  79:   }
  80:   const qs = search.toString()
  81:   return qs ? `?${qs}` : ''
  82: }
  83: 
  84: interface RequestOptions {
  85:   method?: 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE'
  86:   /** Serialised as JSON unless it is already a BodyInit. */
  87:   body?: unknown
  88:   query?: Record<string, QueryValue>
  89:   /** Send without the Authorization header. */
  90:   anonymous?: boolean
  91:   signal?: AbortSignal
  92: }
  93: 
  94: /** Called on any 401 so the app can bounce to the login screen. */
  95: let onUnauthorized: (() => void) | null = null
  96: 
  97: export function setUnauthorizedHandler(handler: (() => void) | null): void {
  98:   onUnauthorized = handler
  99: }

 101: export async function request<T>(path: string, options: RequestOptions = {}): Promise<T> {
 102:   const { method = 'GET', body, query, anonymous = false, signal } = options
 103: 
 104:   const headers = new Headers({ Accept: 'application/json' })
 105:   const isForm = body instanceof FormData
 106: 
 107:   if (body !== undefined && !isForm) headers.set('Content-Type', 'application/json')
 108: 
 109:   if (!anonymous) {
 110:     const token = getToken()
 111:     if (token) headers.set('Authorization', `Bearer ${token}`)
 112:   }
 113: 
 114:   let response: Response
 115:   try {
 116:     response = await fetch(`${API_BASE}${path}${query ? buildQuery(query) : ''}`, {
 117:       method,
 118:       headers,
 119:       body: body === undefined ? undefined : isForm ? (body as FormData) : JSON.stringify(body),
 120:       signal,
 121:     })
 122:   } catch (cause) {
 123:     if (cause instanceof DOMException && cause.name === 'AbortError') throw cause
 124:     throw new ApiError(0, { message: '鏃犳硶杩炴帴鍒版湇鍔″櫒锛岃妫€鏌ュ悗绔槸鍚﹁繍琛? })
 125:   }
 126: 
 127:   if (response.status === 204) return undefined as T
 128: 
 129:   const text = await response.text()
 130:   let payload: unknown = undefined
 131:   if (text) {
 132:     try {
 133:       payload = JSON.parse(text)
 134:     } catch {
 135:       payload = { message: text.slice(0, 300) }
 136:     }
 137:   }
 138: 
 139:   if (!response.ok) {
 140:     if (response.status === 401 && !anonymous) {
 141:       setToken(null)
 142:       onUnauthorized?.()
 143:     }
 144:     throw new ApiError(response.status, (payload ?? {}) as ApiErrorBody)
 145:   }
 146: 
 147:   return payload as T
 148: }

 154: export const http = {

 156:     request<T>(path, { method: 'GET', query, signal }),
