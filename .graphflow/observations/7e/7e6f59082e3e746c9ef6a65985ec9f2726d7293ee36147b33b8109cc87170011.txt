24: // webHandler answers every request that no API route or uploaded file claimed.
25: //
26: // It has two jobs that look alike but must not be confused: unknown API paths
27: // get a JSON error so the frontend's ApiError machinery works, while unknown

30: 	// files is the frontend tree: either the copy embedded in the binary or a

49: // frontendFS decides what serves the pages.
50: //
51: // The frontend is embedded in the binary, so the normal deployment carries no
52: // dist directory at all. An explicit staticDir still wins when it is usable,
53: // which is what makes it possible to iterate on the frontend against a built
54: // binary; a configured path that is missing or holds no index.html falls back
55: // to the embedded copy rather than disabling pages outright.
56: //
57: // Returning an fs.FS rather than a path keeps the serving code below identical
58: // for both sources -- and sidesteps os.DirFS-style path handling mistakes,
59: // since embed.FS and os.DirFS both reject a path that escapes the root.
60: func frontendFS(staticDir string, logger *log.Helper) fs.FS {
61: 	if staticDir != "" {
62: 		abs, err := filepath.Abs(staticDir)
63: 		switch {
64: 		case err != nil:
65: 			logger.Warnf("staticDir %q is unusable (%v); using the embedded frontend", staticDir, err)
66: 		case !isFile(filepath.Join(abs, "index.html")):
67: 			logger.Warnf("staticDir %q has no index.html; using the embedded frontend", abs)
68: 		default:
69: 			logger.Infof("serving the frontend from %s", abs)
70: 			return os.DirFS(abs)
71: 		}
72: 	}
73: 
74: 	if !web.Built() {
75: 		logger.Warn("this binary carries no frontend and no usable staticDir; serving the API only")
76: 		return nil
77: 	}
78: 	logger.Infof("serving the frontend embedded in this binary")
79: 	return web.Dist()
80: }

123: // servePage returns a real file when one exists and index.html otherwise.
124: func (w *webHandler) servePage(res http.ResponseWriter, req *http.Request) {
125: 	// The leading slash before Clean is what makes this safe: every ".." is
126: 	// resolved against the root, so the result can never escape it.
127: 	relative := strings.TrimPrefix(path.Clean("/"+req.URL.Path), "/")

129: 	if relative != "" {

144: // writeAsset sends one frontend file.
145: //
146: // The modtime is deliberately zero. Embedded files have no timestamp, and a
147: // download directory can be updated under a running server, so claiming a
148: // modification time would only produce wrong 304s. Freshness is expressed
149: // through Cache-Control instead, where the Vite content hashes make it exact.
150: func (w *webHandler) writeAsset(res http.ResponseWriter, req *http.Request, name string, body []byte) {
151: 	setCacheControl(res, name)
152: 	// bytes.Reader is what gives ServeContent the seeker it needs for HEAD and
153: 	// range requests; passing a plain reader would silently drop both.
154: 	http.ServeContent(res, req, path.Base(name), time.Time{}, bytes.NewReader(body))
155: }

157: // setCacheControl splits the frontend into the two halves that want opposite
158: // treatment.
159: func setCacheControl(res http.ResponseWriter, name string) {
160: 	switch {
161: 	case name == "index.html":
162: 		// Never cache the entry point: it is the file that names the hashed
163: 		// bundles, so a stale copy pins the browser to a stale build.
164: 		res.Header().Set("Cache-Control", "no-cache")
165: 	case strings.HasPrefix(name, "assets/"):
166: 		// Vite puts a content hash in every bundle filename: a changed file is
167: 		// a changed URL, so these can be cached forever.
168: 		res.Header().Set("Cache-Control", "public, max-age=31536000, immutable")
169: 	default:
170: 		// Everything else (favicon, fonts, robots.txt) keeps a name across
171: 		// builds, so it gets a bounded lifetime and revalidates after that.
172: 		res.Header().Set("Cache-Control", "public, max-age=3600")
173: 	}
174: }

176: // asset reads one frontend file and remembers it. Holding the bytes is close to
177: // free: the whole embedded tree is already mapped into the process image.
178: func (w *webHandler) asset(name string) ([]byte, bool) {
179: 	if cached, ok := w.assets.Load(name); ok {
180: 		return cached.([]byte), true
181: 	}
182: 	// ValidPath rejects absolute paths and any ".." element, so a request can
183: 	// never address a file outside the frontend tree.
184: 	if !fs.ValidPath(name) {
185: 		return nil, false
186: 	}
187: 
188: 	file, err := w.files.Open(name)
189: 	if err != nil {
190: 		return nil, false
191: 	}
192: 	defer file.Close()
193: 
194: 	info, err := file.Stat()
195: 	if err != nil || info.IsDir() {
196: 		return nil, false
197: 	}
198: 	body, err := io.ReadAll(file)
199: 	if err != nil {
200: 		return nil, false
201: 	}
202: 
203: 	w.assets.Store(name, body)
204: 	return body, true
205: }
