85: async function loadYear(year: number, force = false): Promise<void> {
86:   if (!force && loadedYears.has(year)) return
87: 
88:   const cached = force ? null : readCache(year)
89:   if (cached) {
90:     merge(cached)
91:     loadedYears.add(year)

95:   loading.value = true

98:   const timer = window.setTimeout(() => controller.abort(), 8000)
99:   try {
100:     const response = await fetch(`https://timor.tech/api/holiday/year/${year}?type=Y`, {
101:       signal: controller.signal,
102:     })
103:     if (!response.ok) throw new Error(`HTTP ${response.status}`)
104:     const payload = (await response.json()) as {
105:       code?: number
106:       holiday?: Record<string, { holiday?: boolean; name?: string; date?: string }>
107:     }
108:     if (!payload || payload.code !== 0 || !payload.holiday) throw new Error('unexpected payload')

111:     for (const entry of Object.values(payload.holiday)) {

115:         holiday: entry.holiday === true,

157:   const target = new Date(date.getFullYear(), date.getMonth(), date.getDate())
