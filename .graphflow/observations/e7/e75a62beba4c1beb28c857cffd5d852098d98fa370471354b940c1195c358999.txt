228: // rfc3339 renders a timestamp in the format the frontend parses.
229: func rfc3339(t time.Time) string { return t.UTC().Format(time.RFC3339) }
230: 
231: // rfc3339Ptr renders an optional timestamp.
232: func rfc3339Ptr(t *time.Time) *string {
233: 	if t == nil {
234: 		return nil
235: 	}
236: 	s := rfc3339(*t)
237: 	return &s
238: }
