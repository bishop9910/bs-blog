118: function onScrub(event: Event): void {
119:   const value = Number((event.target as HTMLInputElement).value)
120:   if (Number.isFinite(value)) music.seek(value)
121: }

123: function onVolume(event: Event): void {
124:   const value = Number((event.target as HTMLInputElement).value)
125:   if (!Number.isFinite(value)) return
126:   if (music.muted.value && value > 0) music.toggleMute()
127:   music.setVolume(value)
128: }

130: function onKeydown(event: KeyboardEvent): void {
131:   const target = event.target as HTMLElement | null
132:   // Never hijack typing in the playlist search box.
133:   if (target && (target.tagName === 'INPUT' || target.tagName === 'TEXTAREA' || target.isContentEditable)) {
134:     return
135:   }
136: 
137:   if (event.key === 'Escape' && music.expanded.value) {
138:     music.setExpanded(false)
139:     return
140:   }
141: 
142:   // Space toggles playback once the player has something loaded, matching the
143:   // convention of every other media surface.
144:   if (event.code === 'Space' && hasTrack.value && !event.ctrlKey && !event.metaKey && !event.altKey) {
145:     event.preventDefault()
146:     music.toggle()
147:   }
148: }
