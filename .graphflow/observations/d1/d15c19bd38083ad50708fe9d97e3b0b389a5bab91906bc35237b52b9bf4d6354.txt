195: function findNextRestDay(holidayMap: Record<string, HolidayItem>) {
196:   const todayStr = toLocalISOString(new Date());
197:   
198:   // 1. 收集明确的节假日
199:   let candidates = Object.values(holidayMap).map(h => ({
200:     date: h.date,
201:     name: h.name,
202:     isRest: h.holiday === true
203:   }));
204: 
205:   // 2. 补充未来 60 天的周末
206:   const tempDate = new Date();
207:   for(let i=0; i<60; i++) {
208:     const d = new Date(tempDate);
209:     d.setDate(tempDate.getDate() + i);
210:     const dStr = toLocalISOString(d);
211:     const mmdd = dStr.substring(5);
212:     
213:     // 如果已被定义（是节日或调休），跳过
214:     if (holidayMap[mmdd]) continue;

217:     if (day === 0 || day === 6) {
218:       candidates.push({ date: dStr, name: '周末', isRest: true });
219:     }

223:   const next = candidates.find(c => c.isRest && c.date >= todayStr); // 改为 >= 包括今天

228:     nextRestDay.value = { date: next.date, name: next.name };
229:   }
230: }
