123: // Archive groups published posts by year and month, newest first.
124: func (uc *PostUsecase) Archive(ctx context.Context) (*ArchiveDTO, error) {
125: 	posts, err := uc.store.PublishedPosts(ctx)
126: 	if err != nil {
127: 		return nil, internal("无法加载归档", err)
128: 	}
129: 
130: 	// Posts arrive newest-first, so appending preserves that order in the
131: 	// year and month buckets as well.
132: 	years := make([]ArchiveYearDTO, 0, 4)
133: 	yearIndex := make(map[int]int, 4)
134: 	monthIndex := make(map[[2]int]int, 12)
135: 
136: 	for _, post := range posts {
137: 		if post.PublishedAt == nil {
138: 			continue
139: 		}
140: 		published := post.PublishedAt.UTC()
141: 		year, month := published.Year(), int(published.Month())
142: 
143: 		yi, ok := yearIndex[year]
144: 		if !ok {
145: 			years = append(years, ArchiveYearDTO{Year: year, Months: []ArchiveMonthDTO{}})
146: 			yi = len(years) - 1
147: 			yearIndex[year] = yi
148: 		}
149: 
150: 		key := [2]int{year, month}
151: 		mi, ok := monthIndex[key]
152: 		if !ok {
153: 			years[yi].Months = append(years[yi].Months, ArchiveMonthDTO{Month: month, Posts: []PostSummaryDTO{}})
154: 			mi = len(years[yi].Months) - 1
155: 			monthIndex[key] = mi
156: 		}
157: 
158: 		years[yi].Months[mi].Posts = append(years[yi].Months[mi].Posts, toPostSummaryDTO(post))
159: 		years[yi].Count++
160: 	}
161: 
162: 	return &ArchiveDTO{Years: years}, nil
163: }

325: func validateTitle(raw string) (string, error) {
326: 	title := strings.TrimSpace(raw)
327: 	if title == "" {
328: 		return "", errBadRequest("标题不能为空")
329: 	}
330: 	if len([]rune(title)) > maxTitleLength {
331: 		return "", errBadRequest("标题过长")

336: // resolveSlug decides the stored slug.
337: //
338: // A slug the author typed is honoured, and a collision is reported rather than
339: // silently renamed — the editor already checks availability, so a conflict here
340: // means something changed underneath them. A slug derived from the title is
341: // allowed to auto-suffix, because the author never asked for that exact string.
342: func (uc *PostUsecase) resolveSlug(ctx context.Context, requested, title string, excludeID int64) (string, error) {
343: 	explicit := strings.TrimSpace(requested)
344: 	if explicit != "" {
345: 		slug := truncateRunes(textutil.Slugify(explicit), maxSlugLength)
346: 		if slug == "" {
347: 			return "", errBadRequest("slug 只能包含字母、数字或汉字")
348: 		}
349: 		taken, err := uc.store.SlugTaken(ctx, slug, excludeID)
350: 		if err != nil {
351: 			return "", internal("无法检查 slug", err)
352: 		}
353: 		if taken {
354: 			return "", errConflict("该 slug 已被其他文章占用")
355: 		}
356: 		return slug, nil
357: 	}
358: 
359: 	base := truncateRunes(textutil.Slugify(title), maxSlugLength)
360: 	if base == "" {
361: 		base = fmt.Sprintf("post-%d", time.Now().UTC().Unix())
362: 	}
363: 
364: 	candidate := base
365: 	for attempt := 2; attempt <= 100; attempt++ {
366: 		taken, err := uc.store.SlugTaken(ctx, candidate, excludeID)
367: 		if err != nil {
368: 			return "", internal("无法检查 slug", err)
369: 		}
370: 		if !taken {
371: 			return candidate, nil
372: 		}
373: 		candidate = truncateRunes(fmt.Sprintf("%s-%d", base, attempt), maxSlugLength)
374: 	}
375: 
376: 	return "", errConflict("无法为该标题生成唯一 slug，请手动指定")
377: }
