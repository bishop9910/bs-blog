147: // rather than in a router filter. It is deliberately explicit: a new protected

160: func (h *Handler) currentUser(ctx khttp.Context) (*biz.CurrentUser, error) {
161: 	raw := strings.TrimSpace(ctx.Request().Header.Get("Authorization"))
162: 	token := raw
163: 	if len(raw) >= 7 && strings.EqualFold(raw[:7], "bearer ") {
164: 		token = strings.TrimSpace(raw[7:])

166: 	if token == "" {
167: 		return nil, kerrors.Unauthorized("UNAUTHORIZED", "请先登录")

176: // decodeBody reads a JSON request body.
177: //
178: // Bind is not used because its error mapping is coarse, and an oversized body
179: // has to be distinguishable from malformed JSON.
180: func decodeBody(ctx khttp.Context, target any) error {
181: 	body := ctx.Request().Body
182: 	if body == nil {
183: 		return kerrors.BadRequest("INVALID_ARGUMENT", "请求内容为空")
184: 	}
185: 
186: 	if err := json.NewDecoder(body).Decode(target); err != nil {
187: 		var maxErr *http.MaxBytesError
188: 		if errors.As(err, &maxErr) {
189: 			return kerrors.New(http.StatusRequestEntityTooLarge, "PAYLOAD_TOO_LARGE", "请求内容过大")
190: 		}
191: 		if errors.Is(err, io.EOF) {
192: 			return kerrors.BadRequest("INVALID_ARGUMENT", "请求内容为空")
193: 		}
194: 		return kerrors.BadRequest("INVALID_ARGUMENT", "请求内容格式不正确")
195: 	}
196: 	return nil
197: }

252: func (h *Handler) recoverPanic(next http.Handler) http.Handler {
253: 	return http.HandlerFunc(func(res http.ResponseWriter, req *http.Request) {
254: 		defer func() {
255: 			recovered := recover()
256: 			if recovered == nil {
257: 				return
258: 			}
259: 			// net/http raises this sentinel on purpose to abort a response;
260: 			// swallowing it would hide an intentional shutdown.
261: 			if err, ok := recovered.(error); ok && errors.Is(err, http.ErrAbortHandler) {
262: 				panic(recovered)
263: 			}
264: 			h.logger.Errorf("panic serving %s %s: %v\n%s", req.Method, req.URL.Path, recovered, debug.Stack())
265: 			khttp.DefaultErrorEncoder(res, req, kerrors.InternalServer("INTERNAL_ERROR", "服务器内部错误"))
266: 		}()
267: 		next.ServeHTTP(res, req)
268: 	})
269: }

271: func (h *Handler) cors(next http.Handler) http.Handler {
272: 	return http.HandlerFunc(func(res http.ResponseWriter, req *http.Request) {
273: 		origin := strings.TrimSpace(req.Header.Get("Origin"))
274: 		if origin != "" && h.originAllowed(origin) {
275: 			header := res.Header()
276: 			header.Set("Access-Control-Allow-Origin", origin)
277: 			header.Add("Vary", "Origin")
278: 			header.Set("Access-Control-Allow-Methods", "GET, POST, PUT, PATCH, DELETE, OPTIONS")
279: 			header.Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
280: 			header.Set("Access-Control-Max-Age", "600")
281: 		}
282: 		next.ServeHTTP(res, req)
283: 	})
284: }

294: func (h *Handler) logRequest(next http.Handler) http.Handler {
295: 	return http.HandlerFunc(func(res http.ResponseWriter, req *http.Request) {
296: 		started := time.Now()
297: 		tracked := &statusWriter{ResponseWriter: res}
298: 		next.ServeHTTP(tracked, req)
299: 
300: 		message := fmt.Sprintf("%s %s %d %s", req.Method, req.URL.Path, tracked.status(), time.Since(started).Round(time.Millisecond))
301: 		if tracked.status() >= http.StatusInternalServerError {
302: 			h.logger.Error(message)
303: 			return
304: 		}
305: 		h.logger.Info(message)
306: 	})
307: }

313: func (h *Handler) limitBody(next http.Handler) http.Handler {
314: 	return http.HandlerFunc(func(res http.ResponseWriter, req *http.Request) {
315: 		if req.Body != nil && h.maxBody > 0 {
316: 			req.Body = http.MaxBytesReader(res, req.Body, h.maxBody)
