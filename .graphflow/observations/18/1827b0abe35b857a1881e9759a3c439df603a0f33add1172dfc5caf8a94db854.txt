364: // watchLoop drains fsnotify events and debounces them into a full rescan. The
365: // debounce collapses the burst of Create/Write events a single file copy
366: // produces into one rescan after the dust settles.
367: func (r *musicRepo) watchLoop(ctx context.Context, w *fsnotify.Watcher) {
368: 	var timer *time.Timer
369: 	var timerC <-chan time.Time
370: 
371: 	trigger := func() {
372: 		if timer == nil {
373: 			timer = time.NewTimer(watchDebounce)
374: 			timerC = timer.C
375: 			return
376: 		}
377: 		if !timer.Stop() {
378: 			select {
379: 			case <-timer.C:
380: 			default:
381: 			}
382: 		}
383: 		timer.Reset(watchDebounce)
384: 	}
385: 
386: 	for {
387: 		select {
388: 		case <-ctx.Done():
389: 			return
390: 		case err, ok := <-w.Errors:
391: 			if !ok {
392: 				return
393: 			}
394: 			log.Warn("music watch error", "err", err)
395: 		case evt, ok := <-w.Events:
396: 			if !ok {
397: 				return
398: 			}
399: 			if evt.Op&(fsnotify.Create|fsnotify.Write|fsnotify.Remove|fsnotify.Rename) == 0 {
400: 				continue
401: 			}
402: 			trigger()
403: 		case <-timerC:
404: 			timer = nil
405: 			timerC = nil
406: 			if err := r.scan(); err != nil {
407: 				log.Warn("music library rescan issue", "err", err)
408: 			}
409: 			// Cover directories created or removed since the last sync.
410: 			_ = r.addDirWatches(w)
411: 		}
412: 	}
413: }

515: // hashID derives a stable, URL-safe id from an arbitrary string (a relative
516: // file path, an artist name, or an album key). Hashing keeps ids opaque and
517: // immune to path traversal: ids are always resolved through the in-memory index.

519: 	sum := sha256.Sum256([]byte(s))

523: // readLyrics loads a sidecar .lrc file, capped to avoid ballooning memory.
524: func readLyrics(path string) string {
525: 	f, err := os.Open(path)
526: 	if err != nil {
527: 		return ""
528: 	}
529: 	defer f.Close()
530: 	b, err := io.ReadAll(io.LimitReader(f, 1<<20))
531: 	if err != nil {

533: 	}

537: // decodeLyrics normalizes lyric text to UTF-8. Chinese .lrc files are commonly
538: // encoded in GBK/GB2312 (or carry a UTF-8 BOM) rather than UTF-8; serving those
539: // bytes as-is produces mojibake in the JSON output. We keep valid UTF-8 as-is
540: // and fall back to GB18030 (a superset of GBK/GB2312) otherwise.
541: func decodeLyrics(b []byte) string {
542: 	b = bytes.TrimPrefix(b, []byte{0xEF, 0xBB, 0xBF}) // UTF-8 BOM
543: 	if utf8.Valid(b) {
544: 		return string(b)
545: 	}
546: 	if out, err := simplifiedchinese.GB18030.NewDecoder().Bytes(b); err == nil {
547: 		return string(out)
548: 	}
549: 	return string(b)
