75: // scoped builds the shared WHERE clause for a filter.
76: func (f PostFilter) scoped(ctx context.Context, db *gorm.DB) *gorm.DB {
77: 	q := db.WithContext(ctx).Model(&model.Post{})
78: 
79: 	if f.PublishedOnly {
80: 		q = q.Where("status = ? AND visibility = ?", model.StatusPublished, model.VisibilityPublic)
81: 	} else if f.Status == model.StatusDraft || f.Status == model.StatusPublished {
82: 		q = q.Where("status = ?", f.Status)
83: 	}
84: 
85: 	if f.TagSlug != "" {
86: 		q = q.Where("id IN (?)",
87: 			db.WithContext(ctx).Table("post_tags").
88: 				Select("post_tags.post_id").
89: 				Joins("JOIN tags ON tags.id = post_tags.tag_id").
90: 				Where("tags.slug = ?", f.TagSlug),
91: 		)
92: 	}
93: 
94: 	if keyword := strings.TrimSpace(f.Keyword); keyword != "" {
95: 		// Escape LIKE wildcards so a literal % does not match everything.
96: 		escaped := strings.NewReplacer(`\`, `\\`, `%`, `\%`, `_`, `\_`).Replace(keyword)
97: 		like := "%" + escaped + "%"
98: 		q = q.Where(
99: 			`(title LIKE ? ESCAPE '\' OR summary LIKE ? ESCAPE '\' OR content_html LIKE ? ESCAPE '\')`,
100: 			like, like, like,
101: 		)
102: 	}
103: 
104: 	return q

107: // ListPosts returns one page of posts plus the total matching the filter.
108: func (s *Store) ListPosts(ctx context.Context, f PostFilter) ([]model.Post, int64, error) {
109: 	f.Normalize()
110: 
111: 	var total int64
112: 	if err := f.scoped(ctx, s.db).Count(&total).Error; err != nil {
113: 		return nil, 0, fmt.Errorf("data: counting posts: %w", err)
114: 	}
115: 
116: 	column := sortColumns[f.Sort]
117: 	direction := strings.ToUpper(f.Order)
118: 
119: 	// Drafts have a NULL published_at. SQLite sorts NULLs first ascending, so
120: 	// order by nullness first to keep drafts consistently at one end.
121: 	order := column + " " + direction + ", id DESC"
122: 	if column == "published_at" {
123: 		order = "published_at IS NULL, " + order
124: 	}
125: 
126: 	posts := make([]model.Post, 0)
127: 	err := f.scoped(ctx, s.db).
128: 		Preload("Tags").
129: 		Order(order).
130: 		Offset((f.Page - 1) * f.PageSize).
131: 		Limit(f.PageSize).
132: 		Find(&posts).Error
133: 	if err != nil {
134: 		return nil, 0, fmt.Errorf("data: listing posts: %w", err)
135: 	}
136: 	return posts, total, nil

309: // AdjacentPosts finds the neighbours of a post for previous/next navigation.
310: // "Previous" is the next-newest post older than this one; "next" is the
311: // next-oldest post newer than it. Either may be nil at the ends of the feed.
312: func (s *Store) AdjacentPosts(ctx context.Context, publishedAt time.Time, id int64) (prev, next *model.Post, err error) {
313: 	base := func() *gorm.DB {
314: 		return s.db.WithContext(ctx).
315: 			Where("status = ? AND visibility = ? AND published_at IS NOT NULL",
316: 				model.StatusPublished, model.VisibilityPublic)
317: 	}
318: 
319: 	var older model.Post
320: 	err = base().Where("(published_at < ?) OR (published_at = ? AND id < ?)", publishedAt, publishedAt, id).
321: 		Order("published_at DESC, id DESC").First(&older).Error
322: 	switch {
323: 	case err == nil:
324: 		prev = &older
325: 	case errors.Is(err, gorm.ErrRecordNotFound):
326: 		// No older post; prev stays nil.
327: 	default:
328: 		return nil, nil, fmt.Errorf("data: finding previous post: %w", err)
329: 	}
330: 
331: 	var newer model.Post
332: 	err = base().Where("(published_at > ?) OR (published_at = ? AND id > ?)", publishedAt, publishedAt, id).
333: 		Order("published_at ASC, id ASC").First(&newer).Error
334: 	switch {
335: 	case err == nil:
336: 		next = &newer
337: 	case errors.Is(err, gorm.ErrRecordNotFound):
338: 		// No newer post; next stays nil.
339: 	default:

341: 	}

368: 		return stats, fmt.Errorf("data: summing words: %w", err)

372: 	// boundary rows instead and read their timestamps as proper time.Time.
373: 	var first, last model.Post
374: 	if err := publicPosts().Where("published_at IS NOT NULL").
375: 		Order("published_at ASC").Limit(1).Find(&first).Error; err != nil {
376: 		return stats, fmt.Errorf("data: finding first post: %w", err)
377: 	}
378: 	if err := publicPosts().Where("published_at IS NOT NULL").
379: 		Order("published_at DESC").Limit(1).Find(&last).Error; err != nil {
380: 		return stats, fmt.Errorf("data: finding last post: %w", err)
381: 	}
382: 	stats.First = first.PublishedAt
383: 	stats.Last = last.PublishedAt
384: 
385: 	err := s.db.WithContext(ctx).Model(&model.Tag{}).
386: 		Where("id IN (?)",
387: 			s.db.WithContext(ctx).Table("post_tags").Select("post_tags.tag_id").
388: 				Joins("JOIN posts ON posts.id = post_tags.post_id").
389: 				Where("posts.status = ? AND posts.visibility = ?", model.StatusPublished, model.VisibilityPublic),

399: // resolveTags maps tag names onto rows, creating any that do not exist yet.
400: // Deduplication is by slug so "Vue" and "vue" collapse into one tag.
401: func resolveTags(ctx context.Context, tx *gorm.DB, names []string) ([]model.Tag, error) {
402: 	tags := make([]model.Tag, 0, len(names))
403: 	seen := make(map[string]struct{}, len(names))
404: 
405: 	for _, raw := range names {
406: 		name := strings.TrimSpace(raw)
407: 		if name == "" {
408: 			continue
409: 		}
410: 		slug := textutil.Slugify(name)
411: 		if slug == "" {
412: 			continue
413: 		}
414: 		if _, dup := seen[slug]; dup {
415: 			continue
416: 		}
417: 		seen[slug] = struct{}{}
418: 
419: 		var tag model.Tag
420: 		err := tx.WithContext(ctx).Where("slug = ?", slug).First(&tag).Error
421: 		switch {
422: 		case err == nil:
423: 			// Existing tag; reuse it.
424: 		case errors.Is(err, gorm.ErrRecordNotFound):
425: 			tag = model.Tag{Name: name, Slug: slug}
426: 			if err := tx.WithContext(ctx).Create(&tag).Error; err != nil {
427: 				return nil, fmt.Errorf("data: creating tag %q: %w", name, err)
428: 			}
429: 		default:
430: 			return nil, fmt.Errorf("data: looking up tag %q: %w", name, err)
431: 		}
432: 		tags = append(tags, tag)
433: 	}
434: 
435: 	return tags, nil
